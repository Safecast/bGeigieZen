199void LogViewerScreen::upload_detail(const LocalStorage& config) {
200 File log_file = SDInterface::i().get_file(_detail_log_file_path);
201
202 if (!log_file) {
203 M5_LOGD("Failed to open log file!");
204 return;
205 }
206
207 char url[100];
208 sprintf(url, "%s?api_key=%s&", API_LOGFILE_ENDPOINT, config.get_api_key());
209
210 ChunkedHTTPClient http;
211 WiFiClient client;
212
213 // Specify destination for HTTP request
214 if (!http.begin(client, url)) {
215 M5_LOGD("Unable to begin URL connection");
216 http.end(); // Free resources
217 return;
218 }
219
220 // Define boundary and headers
221 char boundary[32];
222 sprintf(boundary, "----WebKitFormBoundaryZen%d", config.get_device_id());
223 char content_type_header[150];
224 sprintf(content_type_header, "multipart/form-data; boundary=%s", boundary);
225
226 http.setUserAgent(HEADER_API_USER_AGENT);
227 http.addHeader("Host", TTSERVE_HOST);
228 http.addHeader("Content-Type", content_type_header);
229 http.addHeader("Transfer-Encoding", "chunked");
230
231 char chunk_header[16];
232 char chunk_buffer[512];
233 size_t chunk_length;
234
235
236 if (http.sendPOSTHeaders()) {
237 // 1. Send the opening boundary and metadata for the file part as a chunk
238 sprintf(chunk_buffer,
239 "--%s\r\n"
240 "Content-Disposition: form-data; name=\"bgeigie_import[description]\"\r\n"
241 "Uploaded from Zen\r\n\r\n"
242 "--%s\r\n"
243 "Content-Disposition: form-data; name=\"bgeigie_import[source]\"; filename=\"%s\"\r\n"
244 "Content-Type: text/plain\r\n\r\n",
245 boundary, boundary, "2024-11-30_1227.log");
246
247 // Calculate header length
248 chunk_length = strlen(chunk_buffer);
249 sprintf(chunk_header, "%X\r\n", chunk_length); // Chunk size in hexadecimal
250 client.print(chunk_header); // Send chunk size
251 client.print(chunk_buffer); // Send chunk data
252 client.print("\r\n"); // End of chunk
253 M5_LOGD("Header chunk sent (%X):\n%s", chunk_length, chunk_buffer);
254
255 // 2. Stream the file content line by line as chunks
256 while (log_file.available()) {
257 chunk_length = log_file.readBytesUntil('\n', chunk_buffer, 511); // Read a line from the file
258 chunk_buffer[chunk_length] = '\n'; // Add the newline character back
259 chunk_buffer[chunk_length + 1] = '\0'; // Null-terminate the buffer
260
261 sprintf(chunk_header, " %X\r\n", chunk_length + 1); // Chunk size in hexadecimal (include '\n')
262 client.print(chunk_header); // Send chunk size
263 client.print(chunk_buffer); // Send chunk data
264 client.print("\r\n"); // End of chunk
265 }
266
267 // 3. Send the closing boundary as a chunk
268 sprintf(chunk_buffer, "--%s--\r\n", boundary);
269 chunk_length = strlen(chunk_buffer);
270 sprintf(chunk_header, "%X\r\n", chunk_length); // Chunk size in hexadecimal
271 client.print(chunk_header); // Send chunk size
272 client.print(chunk_buffer); // Send chunk data
273 client.print("\r\n"); // End of chunk
274 M5_LOGD("Closing boundary chunk sent (%X):\n%s", chunk_length, chunk_buffer);
275
276 // 4. End the chunked transfer by sending a zero-size chunk
277 client.print("0\r\n\r\n");
278 M5_LOGD("End of chunked transfer");
279
280 // 5. Wait for the server response
281 int statusCode = http.handleResponse();
282 String response = http.getString();
283 M5_LOGD("POST complete, (code %d) response:\n%s", statusCode, response.c_str());
284
285 WiFiWrapper_i.update_active();
286 } else {

288 }
