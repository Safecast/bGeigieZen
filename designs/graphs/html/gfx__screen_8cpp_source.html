59void GFXScreen::set_screen_status(ScreenStatus status) {
60 _screen_status = status;
61 switch (_screen_status) {
62 case e_screen_status_on:
63 setBrightness(LEVEL_BRIGHT);

65 case e_screen_status_dim:
66 setBrightness(LEVEL_DIMMED);

80void GFXScreen::setBrightness(uint8_t lvl) {
81
82 if (lvl == LEVEL_BLANKED) {
83 // Turn screen off
84 M5.Lcd.setBrightness(0);
85#ifdef M5_CORE2
86 M5.Power.Axp192.setDCDC3(false);
87 M5.Axp.SetDCDC3(false);
88 M5.Axp.ScreenBreath(0);
89#elif M5_BASIC
90 M5.Lcd.setBrightness(0);
91#endif
92 } else {
93#ifdef M5_CORE2
94 // Make sure screen is turned on
95 M5.Axp.SetDCDC3(true);
96 // Set brightness
97 M5.Axp.ScreenBreath(lvl);
98#elif M5_BASIC
99 if (lvl == LEVEL_BRIGHT)
100 M5.Lcd.setBrightness(200);
101 if (lvl == LEVEL_DIMMED || lvl == LEVEL_BLANKED)
102 M5.Lcd.setBrightness(1);
103#endif
104 }

129 const auto button2 = workers.worker<ZenButton>(k_worker_button_2);
130 const auto button3 = workers.worker<ZenButton>(k_worker_button_3);
131
132 if (button1->is_fresh() || button2->is_fresh() || button3->is_fresh() || (M5.Touch.isEnabled() && M5.Touch.getCount())) {
133 _last_interaction = millis();
134 }
135
136 // Will be set false when returning from wake up
137 bool handle_input = true;
138
139 switch (_screen_status) {
140 case e_screen_status_on:
141 if (TIMEOUT_PASSED(_settings.get_screen_dim_timeout(), _last_interaction)) {
142 set_screen_status(e_screen_status_dim);
143 }
144 else if (TIMEOUT_PASSED(_settings.get_screen_off_timeout(), _last_interaction)) {
145 set_screen_status(e_screen_status_off);
146 }
147 break;
148 case e_screen_status_dim:
149 if (millis() - _last_interaction < (LCD_REFRESH_RATE + 1000)) {
150 set_screen_status(e_screen_status_on);
151 }
152 else if (TIMEOUT_PASSED(_settings.get_screen_off_timeout(), _last_interaction)) {
153 set_screen_status(e_screen_status_off);
154 }
155 break;
156 case e_screen_status_off:
157 if (millis() - _last_interaction < (LCD_REFRESH_RATE + 1000)) {
158 clear();
159 set_screen_status(e_screen_status_on);
160 handle_input = false;
161 }
162 break;
163 }
164
165 if (_screen_status == e_screen_status_off) {
166 return render_screensaver();
167 }
168
169 BaseScreen* new_screen = nullptr;
170 if (handle_input && workers.any_updates()) {
171 if (_menu->menu_open()) {
172 new_screen = _menu->handle_input(_controller, workers);
173 if (new_screen || !_menu->menu_open()) {
174 // Closed menu
175 M5_LOGD("Menu closed");
176 _menu->leave_screen(_controller);
177 clear();
178 }
179 } else if (_screen) {
180 new_screen = _screen->handle_input(_controller, workers);
181 if (new_screen == _menu) {
182 // opened menu, not a new screen
183 M5_LOGD("Menu opened");
184 clear();
185 _menu->enter_screen(_controller);
186 new_screen = nullptr;
187 } else if (new_screen == &DefaultEntryScreen_i) {
188 // entered the default entry screen, handle it right away, no need to render this
189 new_screen = new_screen->handle_input(_controller, workers);
190 }
191 }
192 if (new_screen && new_screen != _screen) {
193 M5_LOGD("New screen entered: %s", new_screen->get_title());
194 _screen->leave_screen(_controller);
195 _screen = new_screen;
196 clear();
197 _screen->enter_screen(_controller);
198
199 // Check if operational mode and save
200 if (new_screen == &DriveModeScreen_i) {
201 _settings.set_last_mode(LocalStorage::e_operational_mode_drive, false);
202 }
203 if (new_screen == &SurveyModeScreen_i) {
204 _settings.set_last_mode(LocalStorage::e_operational_mode_survey, false);
205 }
206 if (new_screen == &FixedModeScreen_i) {
207 _settings.set_last_mode(LocalStorage::e_operational_mode_fixed, false);
208 }
209 if (new_screen == &SatelliteViewScreen_i) {
210 _settings.set_last_mode(LocalStorage::e_operational_mode_satellite, false);
211 }
212
213 }
214 }
215
216 if (workers.any_updates() || handlers.any_updates() || (millis() - _last_render > LCD_REFRESH_RATE)) {
217 M5.Lcd.startWrite();
218 M5.Lcd.setRotation(3);
219 if (_menu->menu_open()) {
220 _menu->do_render(workers, handlers);
221 } else {
222 _screen->do_render(workers, handlers);
223 }
224
225 if (_screen->has_status_bar()) {
226 // Render message if available on top of bar
227 if (_screen->get_error_message(workers, handlers)) {
228 M5.Lcd.setTextColor(LCD_COLOR_DEFAULT, LCD_COLOR_ERROR);
229 uint16_t text_width = M5.Lcd.drawString(_screen->get_error_message(workers, handlers), 0, 220, &fonts::Font2);
230 M5.Lcd.fillRect(text_width, 200, 320 - text_width, 20, LCD_COLOR_BACKGROUND);
231 } else if (_screen->get_status_message(workers, handlers)) {
232 M5.Lcd.setTextColor(LCD_COLOR_BACKGROUND, LCD_COLOR_DEFAULT);
233 uint16_t text_width = M5.Lcd.drawString(_screen->get_status_message(workers, handlers), 0, 220, &fonts::Font2);
234 M5.Lcd.fillRect(text_width, 200, 320 - text_width, 20, LCD_COLOR_BACKGROUND);
235 } else {
236 M5.Lcd.fillRect(0, 200, 320, 20, LCD_COLOR_BACKGROUND);
237 }
238
239 // Render bottom status bar
240 M5.Lcd.drawLine(0, 220, 320, 220, TFT_WHITE);
241 M5.Lcd.setFont(&fonts::Font0);
242
243 // Screen name
244 M5.Lcd.setTextColor(LCD_COLOR_DEFAULT, LCD_COLOR_BACKGROUND);

246 M5.Lcd.print(_screen->get_title());

250 const auto& battery = workers.worker<BatteryIndicator>(k_worker_battery_indicator)->get_data();

327void GFXScreen::render_screensaver() {
328 if (_settings.get_animated_screensaver() && millis() - _last_render > 75) {
329 M5.Lcd.setRotation(3);
330 if (_saver_x + _saver_x_direction < -2 || _saver_x + _saver_x_direction > 318 - SCREENSAVER_TEXT_LENGTH) {
331 _saver_x_direction *= -1;
332 }
333 if (_saver_y + _saver_y_direction < -2 || _saver_y + _saver_y_direction > 230) {
334 _saver_y_direction *= -1;
335 }
336 _saver_x += _saver_x_direction;
337 _saver_y += _saver_y_direction;
338 _saver.pushSprite(_saver_x, _saver_y, TFT_TRANSPARENT);
339 M5.Lcd.setRotation(1);
340 _last_render = millis();
341 }
